08 · SCRIPT DEEP DIVE

JSONL은 통째로 읽지 않고 한 줄씩 처리

async function readLines(filePath, onRecord) {
  const input = fs.createReadStream(filePath, {
    encoding: 'utf8'
  });
  const lines = readline.createInterface({
    input,
    crlfDelay: Infinity
  });

  for await (const line of lines) {
    if (!line.trim()) continue;
    try {
      await onRecord(JSON.parse(line));
    } catch (error) {
      // 깨진 JSON 한 줄만 건너뛴다.
      if (error instanceof SyntaxError) continue;
      throw error;
    }
  }
}

많은 세션은 80개씩, 선택은 키보드로

const PAGE_SIZE = 80;

function moveSelection(direction) {
  const sessions = filteredSessions();
  let index = sessions.findIndex(
    (item) => item.id === state.selectedId
  );

  index = Math.max(0, Math.min(
    sessions.length - 1,
    index + direction
  ));

  // 선택 위치가 표시 범위를 넘으면
  // 다음 80개 묶음까지 렌더링한다.
  if (index >= state.visibleLimit) {
    state.visibleLimit = Math.ceil(
      (index + 1) / PAGE_SIZE
    ) * PAGE_SIZE;
  }
  selectSession(sessions[index].id, true);
}
